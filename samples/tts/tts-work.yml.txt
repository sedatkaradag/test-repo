name: tts-work
# Temporary, branch tts-work only: makes an English narration with Kokoro (free, open source). Writes only branch tts-out.
on:
  push:
    branches: [tts-work]
permissions:
  contents: write
jobs:
  tts:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: {python-version: "3.11"}
      - run: sudo apt-get update -q && sudo apt-get install -y -q espeak-ng
      - run: pip install -q "kokoro>=0.9.4" soundfile
      - run: python tts/make_tts.py bm_george 0.95
      - name: push the narration to branch tts-out
        run: |
          cd out
          git init -q -b tts-out && git config user.name tts && git config user.email tts@users.noreply.github.com
          git add -A && git commit -qm "tts output"
          git push -q -f "https://x-access-token:${{ github.token }}@github.com/${{ github.repository }}" tts-out
